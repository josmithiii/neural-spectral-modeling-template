\begin{slide}[\slideopts,toc={NSMT Overview}]{Neural Spectral Modeling Template (NSMT)}

  \vspace{-1em}

  \begin{itemize}

    \mpitem NSMT is a fork of \emph{Lightning Hydra Template Extended (LHTE)} adapted specifically for
    \textbf{audio spectrograms}

    \mpitem \emph{Key Philosophy:} All ``images'' are audio \emph{spectral representations}:
    \vspace{-0.5em}
    \begin{itemize}
      \mpitem Image \emph{height} = frequency bins
      \mpitem Image \emph{width} = time frames
      \mpitem Image \emph{channels} = alternate spectral representations:
      \begin{itemize}
        \mpitem Alternate time-frequency resolutions
        \mpitem Instantaneous-frequency / group-delay maps
        \mpitem Modulation spectra, Etc.
      \end{itemize}
    \end{itemize}

    \vspace{-0.5em}
    \mpitem \emph{Goal:} Enable \emph{small, accurate, and fast} neural networks for audio processing
    \vspace{-0.5em}
    \begin{itemize}
      \mpitem Accept inductive priors of human hearing
      \mpitem Use efficient CNN architectures with conditioning inputs
      \mpitem Avoid expensive "end-to-end" approaches that require massive data/compute
    \end{itemize}

    \vspace{-0.5em}
    \mpitem Uses only VIMH dataset format from LHTE (drops MNIST/CIFAR support)

    \mpitem Forked from LHTE commit \texttt{3795aad} on August 26, 2024

  \end{itemize}

\end{slide}

\begin{slide}[\slideopts,toc={Spectra}]{Why Spectral Representations?}

  Aren't we supposed to be doing everything \emph{end to end} now?

  Shouldn't our audio inputs be pure \emph{bit streams} by now?
  \begin{itemize}
    \mpitem End-to-end pipelines are feasible but demand \emph{massive} data and computation
    \mpitem Why ``reinvent the ear'' when you know you need an ear?
    \mpitem NSMT adopts inductive priors: the ear is a hardware spectrum analyzer of sorts
    \mpitem A VIMH frame can contain a stack of alternative time-frequency representation ``channels''
    \begin{itemize}
      \mpitem Multiscale planes (different time-frequency resolutions)
      \mpitem Modulation spectra
      \mpitem Spectral metrics over the time-frequency plane
    \end{itemize}
    \mpitem Conditioning (``aux'') inputs bring in measured features (F0, decay-time, pre-emphasis), or outside context
  \end{itemize}

\begin{wideslide}[\slideopts,toc={Context}]{Anchoring in Human Audio Research}
  \twocolumn{
  \vspace{-2em}
  \myFigureRotateToWidth{McDermottKell2018}{-90}{\twidth}{}
  }{
  \vspace{-2em}
  \begin{itemize}
    \mpitem Kell et al. (Neuron 2018) matched human recognition using cochleagrams + CNNs
    \mpitem Cochleagram input was \emph{required} to predict cortical responses
    \mpitem Shared convolutional front-end generalized across speech and music
    \mpitem NSMT adopts spectral presentation like this
    \mpitem NSMT supports multi-resolution stacks, modulation spectra, etc.
    \mpitem NSMT also supports conditioning inputs (e.g., measured features)
    %% \mpitem \emph{Cochleagram} presented as an image input to the convolutional neural network (CNN)
    %% \mpitem For word recognition and music genre classification, this 12-layer network emerged as best (out of 180 tried)
    %% \mpitem The 1st 7 \emph{shared} layers were found to perform comparably with separate 12-layer networks
    %% \mpitem Performs at human level, with similar errors
    %% \mpitem Predicts fMRI responses in the audio cortex
    %% \mpitem Outperforms previous spectrotemporal filter models of the audio cortex
    %% \mpitem Cochleagram required for predicting human responses\\
    %% \mpitem Perhaps the cochleagram can be replaced by vastly more natural data and training (evolution-equivalent)
  \end{itemize}
}

\begin{slide}[\slideopts,toc={Workflow}]{Training \& Experiment Workflow}

  Make targets exist for the following workflow steps:
  \begin{itemize}
    \mpitem \textbf{Synthesize} a VIMH dataset (Hydra config files)
    \mpitem \textbf{Train} a CNN or ViT (specified in Hydra config)
    \mpitem \textbf{Override Hydra Config Files} e.g.,\\ \; \texttt{python src/train.py model=cnn\_small\_ordinal trainer.max\_epochs=50}
    \mpitem \textbf{Output} Lightning checkpoints with config snapshots in run directories\\
    View with TensorBoard et al.
    \mpitem \textbf{Eval} via \texttt{src/eval.py} (same as in LHT and LHTE), or \texttt{src/audio\_reconstruction\_eval.py}
    % \mpitem \textbf{Diagram}  diagrams via \texttt{make td}; TensorBoard via \texttt{make tensorboard}
    % \mpitem \textbf{Testing loop:} \texttt{pytest -k "not slow"} pre-commit, \texttt{make test-all} for releases
  \end{itemize}
\end{slide}

  \vspace{-0.75em}
  \begin{itemize}
    \mpitem \emph{Discrete logits over 0--255 bins} (classic classification):
    \begin{itemize}
      \mpitem \texttt{CrossEntropyLoss}: legacy baseline
      \mpitem \texttt{WeightedCrossEntropyLoss}: adds $|\text{bin\_distance}|^\alpha$ penalty
      \mpitem \texttt{SoftTargetLoss}: KL divergence to smoothed ordinal targets:\\
                      (triangular / gaussian / log-triangular)
    \end{itemize}
    \mpitem \emph{Ordinal logits with perceptual regression readout}:
    \begin{itemize}
      \mpitem \texttt{OrdinalRegressionLoss}: softmax $\rightarrow$ weighted-average prediction\\
      Loss reported in perceptual units
      \mpitem \texttt{QuantizedRegressionLoss}: scalar output over quantized steps \\
      Scales L1/L2/Huber by the dataset step size
    \end{itemize}
    \mpitem \emph{Continuous heads} (true regression mode):
    \begin{itemize}
      \mpitem \texttt{NormalizedRegressionLoss}: sigmoid outputs in $[0,1]$,\\
      auto-denormalized to perceptual units (current default)
    \end{itemize}
